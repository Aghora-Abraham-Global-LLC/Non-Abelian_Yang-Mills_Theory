\label{sec:introduction}

In the foundational formulation of quantum gauge field theory, non-Abelian Yang--Mills theory in four Euclidean dimensions represents the central junction connecting differential geometry, topological quantum field theory, and particle physics~\cite{YangMills1954, Donaldson1983}. Let $G$ be a compact, simple Lie group (such as $\mathrm{SU}(N)$) with Lie algebra $\mathfrak{g}$, and let $P \to M$ be a principal $G$-bundle over an oriented, smooth, compact Riemannian 4-manifold $(M, g)$.

Building upon the foundations established across the \emph{Purist Monograph Series} (Vol.~I~\cite{PuristVol1Monograph2026}, Vol.~II~\cite{PuristVol2Monograph2026}, Vol.~III~\cite{PuristVol3Monograph2026}, and Vol.~IV~\cite{RelativisticThreeBody2026}), alongside our certified engineering lineage across \emph{Automata V10.0}~\cite{AutomataV10Monograph2026}, \emph{Finite-Dimensional V10.0}~\cite{FiniteV10Monograph2026}, \emph{Nath Darsana V2.0}~\cite{NathDarsanaV2Monograph2026}, \emph{SPDE Version 6.0}~\cite{SPDEV60}, and \emph{Pillar X Sovereign Capstone}~\cite{PillarXMonograph2026}, this monograph constructs the complete canonical solution on the 22-dimensional composite fiber bundle:
\begin{equation}
  \Eym \coloneqq \Mbf \times \mathbb{E}^8.
  \label{eq:eym_22d_def}
\end{equation}

A gauge connection is represented by a connection 1-form $A \in \Omega^1(M, \mathfrak{g})$. The associated field strength (curvature 2-form) is defined by the Cartan structural equation:
\begin{equation}
  F_A = \diff A + \frac{1}{2} [A \wedge A] \in \Omega^2(M, \mathfrak{g}).
  \label{eq:cartan_curvature}
\end{equation}
Under gauge transformations $u \in \mathcal{G} = \operatorname{Aut}(P)$, the curvature transforms covariantly: $F_A \mapsto u^{-1} F_A u$. The classical Yang--Mills action is the normalized $L^2$-norm of the curvature:
\begin{equation}
  S_{\mathrm{YM}}(A) = \frac{1}{g_{\mathrm{YM}}^2} \int_M -\Tr(F_A \wedge *F_A) = \frac{1}{g_{\mathrm{YM}}^2} \|F_A\|_{L^2}^2,
  \label{eq:ym_action}
\end{equation}
where $*: \Omega^k(M) \to \Omega^{4-k}(M)$ is the Hodge star operator induced by the Riemannian metric $g_{\mu\nu}$, and $\Tr$ denotes the invariant negative-definite Cartan--Killing form normalized such that $\Tr(T^a T^b) = -\frac{1}{2}\delta^{ab}$, ensuring $\|F_A\|_{L^2}^2 = \frac{1}{2}\int F_{\mu\nu}^a F^{a\mu\nu}\sqrt{g}\,\diff^4 x \ge 0$.

\subsection{The Hodge Star Involution as an Invariant Axis Archetype}
\label{subsec:hodge_star_archetype}

On middle-dimensional forms ($k = 2$) in four Riemannian dimensions, the Hodge star operator satisfies the involutive property:
\begin{equation}
  *^2 = (-1)^{k(4-k)} \mathrm{id} = (-1)^{2(2)} \mathrm{id} = +\mathrm{id}_{\Omega^2}.
  \label{eq:hodge_involution}
\end{equation}
Consequently, the Hodge star acts as an algebraic reflection involution on $\Omega^2(M, \mathfrak{g})$, inducing an exact eigenspace decomposition into self-dual (SD, $* = +1$) and anti-self-dual (ASD, $* = -1$) sub-bundles:
\begin{equation}
  \Omega^2(M, \mathfrak{g}) = \Omega^2_+(M, \mathfrak{g}) \oplus \Omega^2_-(M, \mathfrak{g}),
  \label{eq:hodge_orthogonal_decomposition}
\end{equation}
with projection operators:
\begin{equation}
  P_{\pm} \coloneqq \frac{1}{2} (\mathrm{id} \pm *), \qquad P_+ + P_- = \mathrm{id}, \quad P_\pm^2 = P_\pm, \quad P_+ P_- = 0.
\end{equation}
Decomposing the curvature 2-form into its self-dual and anti-self-dual projections:
\begin{equation}
  F_A = F_A^+ + F_A^-, \qquad F_A^\pm \coloneqq P_\pm F_A = \frac{1}{2} (F_A \pm *F_A),
\end{equation}
we observe that $F_A^+$ and $F_A^-$ are pointwise orthogonal with respect to the Riemannian metric: $\Tr(F_A^+ \wedge *F_A^-) = -\Tr(F_A^+ \wedge F_A^-) \equiv 0$.

\subsubsection{The Bogomol'nyi Bound and the Invariant Axis Leaf}
The second Chern class (instanton topological invariant) of the principal bundle $P$ is given by the integral:
\begin{equation}
  k \coloneqq c_2(P) = \frac{1}{8\pi^2} \int_M -\Tr(F_A \wedge F_A) = \frac{1}{8\pi^2} \left( \|F_A^+\|_{L^2}^2 - \|F_A^-\|_{L^2}^2 \right) \in \mathbb{Z}.
  \label{eq:topological_charge}
\end{equation}
Combining Eq.~\eqref{eq:ym_action} and Eq.~\eqref{eq:topological_charge}, the classical Yang--Mills action can be expressed identically as:
\begin{align}
  S_{\mathrm{YM}}(A) &= \frac{1}{g_{\mathrm{YM}}^2} \left( \|F_A^+\|_{L^2}^2 + \|F_A^-\|_{L^2}^2 \right) \nonumber \\
  &= \frac{8\pi^2 |k|}{g_{\mathrm{YM}}^2} + \frac{2}{g_{\mathrm{YM}}^2} \|F_A^-\|_{L^2}^2 \quad (\text{for } k \ge 0).
  \label{eq:bogomolnyi_identity}
\end{align}
Because $\|F_A^-\|_{L^2}^2 \ge 0$, the action satisfies the strict Bogomol'nyi inequality:
\begin{equation}
  S_{\mathrm{YM}}(A) \ge \frac{8\pi^2 |k|}{g_{\mathrm{YM}}^2},
  \label{eq:bogomolnyi_bound}
\end{equation}
with equality holding if and only if the connection satisfies the first-order self-dual instanton equation:
\begin{equation}
  F_A^- \equiv 0 \iff F_A = *F_A.
  \label{eq:self_dual_equation}
\end{equation}

\begin{theorem}[\texttt{THM-YM-01}: Vanishing Transverse Action Gradient on the Self-Dual Leaf]
\label{thm:transverse_gradient_vanishing}
Let $\mathcal{A}_k$ denote the affine space of smooth connections on $P$ with topological charge $k \ge 0$, and let $\mathcal{M}_k \coloneqq \{ [A] \in \mathcal{A}_k / \mathcal{G} \mid F_A^- = 0 \}$ denote the self-dual instanton moduli space. 

Along the self-dual leaf $\mathcal{M}_k$, the first variation of the classical Yang--Mills action vanishes identically:
\begin{equation}
  \left. \frac{\delta S_{\mathrm{YM}}}{\delta A} \right|_{F_A^- = 0} \equiv \mathbf{0}.
\end{equation}
Furthermore, the transverse Hessian operator is strictly positive semi-definite on $\operatorname{ker}(D_A^*) \cap \Omega^1(M, \mathfrak{g})$:
\begin{equation}
  \left. \frac{\delta^2 S_{\mathrm{YM}}}{\delta A^2} \right|_{F_A^- = 0} = D_A^* D_A + 2 [F_A^+, \cdot] \succeq 0,
\end{equation}
proving that the instanton moduli space $\mathcal{M}_k$ constitutes the exact gauge-theoretic realization of the unperturbed invariant axis archetype (\emph{madhya-dh\={a}man}), where all transverse, non-self-dual restoring forces vanish identically.
\end{theorem}

\begin{proof}
Taking an arbitrary smooth variational perturbation $\delta A = a \in \Omega^1(M, \mathfrak{g})$, the first variation of the action is a 4-form pairing:
\begin{equation}
  \delta S_{\mathrm{YM}}(A) = \frac{2}{g_{\mathrm{YM}}^2} \int_M -\Tr\left( a \wedge D_A *F_A \right) = \frac{2}{g_{\mathrm{YM}}^2} \langle a, D_A^* F_A \rangle_{L^2}.
\end{equation}
On a Riemannian 4-manifold, the formal adjoint of the covariant exterior derivative on 2-forms is $D_A^* = -* D_A *$. Decomposing the curvature into its self-dual and anti-self-dual components:
\begin{equation}
  D_A^* F_A = -* D_A (*F_A) = -* D_A (F_A^+ - F_A^-) = -* D_A F_A^+ + * D_A F_A^-.
\end{equation}
By the differential Bianchi identity, $D_A F_A \equiv D_A F_A^+ + D_A F_A^- \equiv 0$, which implies $D_A F_A^+ = -D_A F_A^-$. Substituting this identity yields:
\begin{equation}
  D_A^* F_A = -* (-D_A F_A^-) + * D_A F_A^- = * D_A F_A^- + * D_A F_A^- = 2 * D_A F_A^-.
\end{equation}
Evaluating this on any self-dual connection where $F_A^- \equiv 0$:
\begin{equation}
  \left. D_A^* F_A \right|_{F_A^- = 0} = 2 * D_A (\mathbf{0}) \equiv \mathbf{0},
